cmake_minimum_required(VERSION 3.16)
project(common_golf VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CGP_DEPENDENCIES "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Verified dependency source directory")
option(CGP_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(CGP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(cmake/wolfssl.cmake)
add_library(wolfssl STATIC ${WOLFSSL_SOURCES})
target_include_directories(wolfssl PUBLIC ${WOLFSSL_INCLUDES})
target_compile_definitions(wolfssl PUBLIC WOLFSSL_USER_SETTINGS)
include(cmake/ngtcp2.cmake)
add_library(ngtcp2 STATIC ${NGTCP2_SOURCES})
target_include_directories(ngtcp2 PUBLIC ${NGTCP2_INCLUDES} PRIVATE ${NGTCP2_PRIVATE_INCLUDES})
target_compile_definitions(ngtcp2 PUBLIC NGTCP2_STATICLIB PRIVATE BUILDING_NGTCP2 HAVE_ARPA_INET_H=1 HAVE_NETINET_IN_H=1 HAVE_UNISTD_H=1)
target_link_libraries(ngtcp2 PUBLIC wolfssl)
add_library(cjson STATIC "${CGP_DEPENDENCIES}/cjson/cJSON.c")
target_include_directories(cjson PUBLIC "${CGP_DEPENDENCIES}/cjson")
add_library(common_golf STATIC src/transport.c src/identity.c src/protocol.c src/menu.c src/json_guard.c src/json_extract.c src/framing.c)
add_library(common_golf::common_golf ALIAS common_golf)
target_include_directories(common_golf PUBLIC include PRIVATE src)
target_compile_options(common_golf PRIVATE -Wall -Wextra -Werror)
target_link_libraries(common_golf PUBLIC ngtcp2 cjson m)
add_library(cgp_posix STATIC src/posix.c)
target_include_directories(cgp_posix PUBLIC include)
add_executable(cgp-peer tools/peer.c)
target_link_libraries(cgp-peer PRIVATE common_golf cgp_posix)
add_executable(cgp-demo ../../examples/c/main.c ../../examples/c/demo_app.c)
target_link_libraries(cgp-demo PRIVATE common_golf cgp_posix)
include(CTest)
if(BUILD_TESTING)
foreach(test framing json_guard protocol menu transport draft3 tx_bound)
  add_executable(test_${test} tests/test_${test}.c)
  target_include_directories(test_${test} PRIVATE src)
  target_compile_options(test_${test} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  target_link_libraries(test_${test} PRIVATE common_golf cgp_posix)
  add_test(NAME ${test} COMMAND test_${test})
endforeach()
endif()
find_path(DNSSD_INCLUDE_DIR dns_sd.h)
find_library(DNSSD_LIBRARY NAMES dns_sd)
if(APPLE AND DNSSD_INCLUDE_DIR)
  set(DNSSD_LIBRARY "-lSystem")
endif()
if(DNSSD_INCLUDE_DIR AND DNSSD_LIBRARY)
  add_library(cgp_discovery STATIC src/discovery.c)
  target_include_directories(cgp_discovery PUBLIC include PRIVATE ${DNSSD_INCLUDE_DIR})
  target_link_libraries(cgp_discovery PUBLIC common_golf ${DNSSD_LIBRARY})
  target_link_libraries(cgp-demo PRIVATE cgp_discovery)
  target_compile_definitions(cgp-demo PRIVATE CGP_HAVE_DNSSD=1)
  target_link_libraries(cgp-peer PRIVATE cgp_discovery)
  target_compile_definitions(cgp-peer PRIVATE CGP_HAVE_DNSSD=1)
  install(TARGETS cgp_discovery ARCHIVE DESTINATION lib)
endif()
install(TARGETS common_golf cgp_posix wolfssl ngtcp2 cjson ARCHIVE DESTINATION lib)
install(FILES include/cgp.h include/framing.h include/cgp_discovery.h DESTINATION include)
