cmake_minimum_required(VERSION 3.20)
project(cgp_rust_c_example C)
set(CGP_RUST_LIBRARY "" CACHE FILEPATH "Rust import library (.dll.lib), .dylib or .so")
if(NOT EXISTS "${CGP_RUST_LIBRARY}")
  message(FATAL_ERROR "Build the Rust cdylib and set CGP_RUST_LIBRARY")
endif()
add_executable(cgp-c-ffi main.c)
target_include_directories(cgp-c-ffi PRIVATE ../../implementations/rust/include)
target_link_libraries(cgp-c-ffi PRIVATE "${CGP_RUST_LIBRARY}")
enable_testing()
add_test(NAME abi-smoke COMMAND cgp-c-ffi)
